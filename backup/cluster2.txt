#include "hnswlib/hnswlib.h"
#define BTREE_M 3
#define BTREE_D 2

using namespace hnswlib;

class RangeHNSW {
public:
    struct CompareByFirst {
        constexpr bool operator()(std::pair<float, tableint> const& a,
                                  std::pair<float, tableint> const& b) const noexcept {
            return a.first < b.first;
        }
    };

    typedef std::priority_queue<std::pair<float, tableint>, 
                                std::vector<std::pair<float, tableint>>, 
                                CompareByFirst> ResultHeap;

    RangeHNSW(
            int d,
            size_t eleNum,
            size_t maxEleNum,
            float* vecData,
            int* keyList,
            int* valueList,
            int m,
            int ef_con
    );

    ~RangeHNSW();

    std::priority_queue<std::pair<float, hnswlib::labeltype>> 
    queryRange(float *vecData, int rangeL, int rangeR, int k, int ef_s);

    void addPoint(int key, int value, char* data);
    void erase(int key);
    void resize(size_t newMaxN);

private:
    struct node {
        int entryPoint;
        int keynum;
        int key[BTREE_M];
        struct node* child[BTREE_M + 1];
        short int layer;
        
        node() : entryPoint(-1), keynum(0), layer(0) {
            memset(key, 0, sizeof(key));
            memset(child, 0, sizeof(child));
        }
    };

    struct Cluster {
        std::vector<float> centroid;
        std::vector<tableint> representatives;
        float radius;
        int minAttr, maxAttr;
        int level;
        
        Cluster(int dim) : radius(0), minAttr(INT_MAX), maxAttr(INT_MIN), level(0) {
            centroid.resize(dim, 0);
        }
    };

    // 聚类统计信息
    struct ClusterStats {
        int level;
        size_t numPoints;
        int numClusters;
        float avgClusterSize;
        float buildTime;
        float queryHitRate;
        
        ClusterStats() : level(0), numPoints(0), numClusters(0), 
                        avgClusterSize(0), buildTime(0), queryHitRate(0) {}
    };

    size_t maxNum;
    size_t eleCount;
    size_t sizeLinkList;
    
    node* root;
    char* vecData_;
    int* keyList_;
    int* valueList_;
    bool* isDeleted;
    
    int M;
    int ef_construction;
    int dim;
    int maxLayer;
    int skipLayer;
    
    double mult_;
    double revSize_;
    
    std::vector<char *> linklist;
    std::vector<short int> searchLayer;
    unsigned int *visited_array;
    unsigned int tag;
    std::vector<int> sortedArray;
    std::unordered_map<int, int> key2Id;
    std::mt19937 eng;
    
    hnswlib::L2Space space;
    size_t data_size_;
    DISTFUNC<float> fstdistfunc_;
    void *dist_func_param_;
    
    int numEdges;

    std::vector<std::unique_ptr<Cluster>> clusters;
    std::unordered_map<int, std::vector<int>> levelClusters;
    std::vector<ClusterStats> clusterStats;
    std::unordered_map<int, float> levelHitCount;
    int totalQueries;

    bool cmp(int a, int b);
    int findEntryLayer(int Layer) const;
    int findRight(node* nd);
    void updateEntry(node *nd);
    inline char *getDataByInternalId(tableint internal_id) const;
    linklistsizeint *get_linklist(tableint internal_id, int layer) const;
    unsigned short int getListCount(linklistsizeint *ptr) const;
    void setListCount(linklistsizeint *ptr, unsigned short int size) const;
    std::vector<float> getVector(tableint id) const;
    float distL2(const float* a, const float* b) const;
    float distL2(const std::vector<float>& a, const std::vector<float>& b) const;
    float distL2(const float* a, const std::vector<float>& b) const;

    node* buildTree(int eleNum);
    void traverse(std::vector<tableint>& result, node* nd);
    node* findHighNode(node* node, int rangeL, int rangeR);
    int findPosition(node* nd, int value);
    void collectLevelPoints(node* nd, int targetLevel, std::vector<tableint>& points);

    tableint findEntry(const void *query_data, node *nd, tableint currObj) const;
    ResultHeap searchBaseLayer(const std::vector<tableint> &ep_ids, const void *data_point, int layer);
    ResultHeap searchBaseLayer0(std::vector<tableint> ep_ids, const void *data_point, 
                                 int Layer, int rangeL, int rangeR, int ef, int splitPoint);
    void getNeighborsByHeuristic2(ResultHeap &top_candidates, const size_t M);
    tableint connectEdges(const void *data_point, tableint cur_c, ResultHeap &top_candidates, int layer);

    void addPoint(int id);
    void erase(node *nd, int id);
    void refresh(node *nd, int refreshId);
    void mergeNode(node *nd, int mergeId);
    tableint insert(node* nd, int id);
    void splitNode(node *nd, int splitId);
    void refresh(node *nd);

    void buildClusterIndex();
    void buildClustersForLevel(const std::vector<tableint>& points, int k, int level);
    std::vector<int> kmeansPlusPlusInit(const std::vector<tableint>& points, int k);
    std::vector<tableint> findClusterCandidates(float* queryVec, int rangeL, int rangeR, node* targetNode);
    float calculateUniqueness(const std::vector<tableint>& points);
    void validateClusterIndex();
};

// ==================== 构造函数实现 ====================

RangeHNSW::RangeHNSW(
        int d,
        size_t eleNum,
        size_t maxEleNum,
        float* vecData,
        int* keyList,
        int* valueList,
        int m,
        int ef_con
):
        M(m), ef_construction(ef_con), space(d), dim(d), 
        linklist(maxEleNum), searchLayer(maxEleNum), 
        eleCount(eleNum), maxNum(maxEleNum), numEdges(0),
        mult_(0), revSize_(0), tag(0), totalQueries(0) {

    skipLayer = log(M)/log(BTREE_D);
    maxLayer = floor(log((float)maxEleNum) / log(BTREE_D));

    std::cout << "M=" << M << ", maxLayer=" << maxLayer << ", ef_construction=" << ef_con << std::endl;

    visited_array = new unsigned int[maxEleNum]();
    std::random_device rd;
    eng = std::mt19937(rd());

    data_size_ = space.get_data_size();
    fstdistfunc_ = space.get_dist_func();
    dist_func_param_ = space.get_dist_func_param();

    keyList_ = new int[maxEleNum];
    valueList_ = new int[maxEleNum];
    vecData_ = (char*)(new float[maxEleNum * dim]);
    isDeleted = new bool[maxEleNum]();
    
    memcpy(keyList_, keyList, eleNum * sizeof(int));
    memcpy(valueList_, valueList, eleNum * sizeof(int));
    memcpy(vecData_, vecData, eleNum * dim * sizeof(float));

    std::cout << "Step 1\n";
    
    mult_ = 1 / log(1.0 * M);
    revSize_ = 1.0 / mult_;
    sizeLinkList = (M * sizeof(tableint) + sizeof(linklistsizeint));

    linklist.assign(maxEleNum, nullptr);
    sortedArray.reserve(eleNum);
    key2Id.clear();

    for(int i = 0; i < eleNum; i++) {
        key2Id[keyList[i]] = i;
        linklist[i] = (char *)calloc((maxLayer + 1), sizeLinkList);
        sortedArray.push_back(i);
        
        for(int layer = 0; layer <= maxLayer; layer++) {
            unsigned int *newListData = (unsigned int *)get_linklist(i, layer);
            setListCount(newListData, 0);
        }
    }

    sort(sortedArray.begin(), sortedArray.end(), 
         [this](int a, int b) { return this->cmp(a, b); });

    root = buildTree(eleNum);
    
    if (root) {
        std::cout << "Building cluster index for acceleration..." << std::endl;
        buildClusterIndex();
        validateClusterIndex();
        std::cout << "Cluster index built with " << clusters.size() << " clusters" << std::endl;
    }
}

RangeHNSW::~RangeHNSW() {
    delete[] visited_array;
    delete[] keyList_;
    delete[] valueList_;
    delete[] (float*)vecData_;
    delete[] isDeleted;
    
    for (auto ptr : linklist) {
        if (ptr) free(ptr);
    }
}

// ==================== B树构建 ====================

RangeHNSW::node* RangeHNSW::buildTree(int eleNum) {
    if (eleNum == 0) return nullptr;
    
    std::queue<std::pair<std::pair<int,int>, node*>> q[2];
    int qid = 0;
    
    // 叶子节点层数为0
    for(int i = 0; i < eleNum; i++) {
        node *nd = new node();
        nd->layer = 0;
        nd->entryPoint = sortedArray[i];
        q[qid].push({{i, i}, nd});
    }
    
    int currentLayer = 0;
    std::vector<size_t> layerSizes;
    
    while(q[qid].size() > 1) {
        currentLayer++;
        size_t nodesInLayer = q[qid].size();
        layerSizes.push_back(nodesInLayer);
        
        std::cout << "Building layer " << currentLayer 
                  << " with " << nodesInLayer << " nodes" << std::endl;
        
        int nxtqid = qid ^ 1;
        
        while(!q[qid].empty()) {
            std::vector<std::pair<int,int>> tmp;
            int numChild;
            
            if(q[nxtqid].size() % 2 == 0) {
                numChild = (q[qid].size() >= 2 * BTREE_D) ? BTREE_D : q[qid].size();
            } else {
                if (q[qid].size() >= 2 * BTREE_M) numChild = BTREE_M;
                else if (q[qid].size() <= BTREE_M) numChild = q[qid].size();
                else numChild = q[qid].size() / 2;
            }
            
            if (numChild == 0) numChild = 1;
            
            node* nd = new node();
            nd->keynum = numChild - 1;
            
            for(int i = 0; i < numChild; i++) {
                if (q[qid].empty()) break;
                
                auto t = q[qid].front();
                tmp.push_back(t.first);
                q[qid].pop();
                
                if (i != 0 && t.first.first < (int)sortedArray.size()) {
                    nd->key[i - 1] = sortedArray[t.first.first];
                }
                nd->child[i] = t.second;
            }
            
            if (!tmp.empty()) {
                std::uniform_int_distribution<> distr(0, numChild - 1);
                nd->entryPoint = nd->child[distr(eng)]->entryPoint;
                nd->layer = currentLayer;
                
                // 构建这一层的图
                for(int i = 0; i < numChild; i++) {
                    for (int ii = tmp[i].first; ii <= tmp[i].second; ii++) {
                        int id = sortedArray[ii];
                        ResultHeap candidates;
                        char *data = getDataByInternalId(id);
                        
                        unsigned int *listData = (unsigned int *)get_linklist(id, nd->layer - 1);
                        int size = getListCount(listData);
                        
                        if (size > 0) {
                            tableint *listD = (tableint *)(listData + 1);
                            for (int j = 0; j < size; j++) {
                                if (listD[j] >= 0 && listD[j] < eleCount && !isDeleted[listD[j]]) {
                                    candidates.emplace(
                                        fstdistfunc_(data, getDataByInternalId(listD[j]),
                                                    dist_func_param_), listD[j]);
                                }
                            }
                        }
                        
                        for (int j = 0; j < numChild; j++) {
                            if (i != j && nd->child[j]) {
                                tableint ep_id = findEntry(data, nd->child[j], nd->child[j]->entryPoint);
                                if (ep_id >= 0 && ep_id < eleCount) {
                                    std::vector<tableint> ep_ids = {ep_id};
                                    ResultHeap r = searchBaseLayer(ep_ids, data, nd->layer - 1);
                                    getNeighborsByHeuristic2(r, M);
                                    while (!r.empty()) {
                                        auto pr = r.top();
                                        r.pop();
                                        if (pr.second >= 0 && pr.second < eleCount) {
                                            candidates.push(pr);
                                        }
                                    }
                                }
                            }
                        }
                        
                        getNeighborsByHeuristic2(candidates, M);
                        
                        unsigned int *newListData = (unsigned int *)get_linklist(id, nd->layer);
                        tableint *newListD = (tableint *)(newListData + 1);
                        int indx = 0;
                        while (candidates.size() > 0) {
                            newListD[indx] = candidates.top().second;
                            candidates.pop();
                            indx++;
                        }
                        setListCount(newListData, indx);
                        numEdges += indx;
                    }
                }
                
                q[nxtqid].push({{tmp[0].first, tmp.back().second}, nd});
            } else {
                delete nd;
            }
        }
        
        qid = nxtqid;
    }
    
    node* rootNode = q[qid].front().second;
    
    // 验证层数设置
    std::cout << "\n=== Layer Verification ===" << std::endl;
    std::cout << "Root layer = " << rootNode->layer << std::endl;
    std::cout << "maxLayer (calculated) = " << maxLayer << std::endl;
    std::cout << "Actual tree height = " << currentLayer << std::endl;
    
    for (size_t i = 0; i < layerSizes.size(); i++) {
        std::cout << "Layer " << i+1 << " size: " << layerSizes[i] << std::endl;
    }
    
    std::cout << "Edge num: " << numEdges << std::endl;
    std::cout << "Average edges: " << numEdges*1.0/eleNum << std::endl;
    
    return rootNode;
}

// ==================== 查询函数实现 ====================

std::priority_queue<std::pair<float, hnswlib::labeltype>> 
RangeHNSW::queryRange(float *vecData, int rangeL, int rangeR, int k, int ef_s) {
    if (!root) {
        return std::priority_queue<std::pair<float, hnswlib::labeltype>>();
    }
    
    node* highNode = findHighNode(root, rangeL, rangeR);
    if (!highNode) {
        return std::priority_queue<std::pair<float, hnswlib::labeltype>>();
    }

    std::vector<tableint> clusterCandidates;
    if (!clusters.empty()) {
        clusterCandidates = findClusterCandidates(vecData, rangeL, rangeR, highNode);
    }
    
    std::vector<tableint> ep_ids;
    
    int belongL = highNode->keynum;
    int belongR = highNode->keynum;
    
    for(int i = 0; i < highNode->keynum; i++) {
        if (highNode->key[i] >= 0 && highNode->key[i] < (int)eleCount) {
            if(rangeL < valueList_[highNode->key[i]] ||
               (rangeL == valueList_[highNode->key[i]] && 
                highNode->child[i] && 
                rangeL == valueList_[findRight(highNode->child[i])])){
                belongL = i;
                break;
            }
        }
    }
    for(int i = 0; i < highNode->keynum; i++) {
        if (highNode->key[i] >= 0 && highNode->key[i] < (int)eleCount) {
            if(rangeR < valueList_[highNode->key[i]]){
                belongR = i;
                break;
            }
        }
    }

    int sp = -1;
    if(belongL == belongR) {
        if (highNode->entryPoint >= 0 && highNode->entryPoint < (int)eleCount) {
            ep_ids.push_back(highNode->entryPoint);
        }
    } else if(belongL == belongR - 1) {
        if (highNode->child[belongL]) {
            node* nodeL = highNode->child[belongL];
            while(nodeL && nodeL->layer != 0 && nodeL->keynum > 0 && 
                  nodeL->child[nodeL->keynum] &&
                  valueList_[nodeL->key[nodeL->keynum - 1]] < rangeL) {
                nodeL = nodeL->child[nodeL->keynum];
            }
            if (nodeL) {
                tableint ep1;
                if (nodeL->layer != 0 && nodeL->child[nodeL->keynum]) {
                    ep1 = findEntry(vecData, nodeL, nodeL->child[nodeL->keynum]->entryPoint);
                } else {
                    ep1 = nodeL->entryPoint;
                }
                if (ep1 >= 0 && ep1 < (int)eleCount) {
                    ep_ids.push_back(ep1);
                    searchLayer[ep1] = nodeL->layer;
                }
                if (belongL < highNode->keynum) {
                    sp = highNode->key[belongL];
                }
            }
        }

        if (highNode->child[belongR]) {
            node* nodeR = highNode->child[belongR];
            while(nodeR && nodeR->layer != 0 && nodeR->keynum > 0 && 
                  nodeR->child[0] &&
                  valueList_[nodeR->key[0]] > rangeR) {
                nodeR = nodeR->child[0];
            }
            if (nodeR) {
                tableint ep2;
                if (nodeR->layer != 0 && nodeR->child[0]) {
                    ep2 = findEntry(vecData, nodeR, nodeR->child[0]->entryPoint);
                } else {
                    ep2 = nodeR->entryPoint;
                }
                if (ep2 >= 0 && ep2 < (int)eleCount) {
                    ep_ids.push_back(ep2);
                    searchLayer[ep2] = nodeR->layer;
                }
            }
        }
    } else {
        if (belongL + 1 <= belongR - 1) {
            std::uniform_int_distribution<> distr(belongL + 1, belongR - 1);
            int childIdx = distr(eng);
            if (highNode->child[childIdx]) {
                tableint high_ep = highNode->child[childIdx]->entryPoint;
                if (high_ep >= 0 && high_ep < (int)eleCount) {
                    tableint ep = findEntry(vecData, highNode, high_ep);
                    if (ep >= 0 && ep < (int)eleCount) {
                        ep_ids.push_back(ep);
                        searchLayer[ep] = highNode->layer;
                    }
                }
            }
        }
    }
    
    for (auto cid : clusterCandidates) {
        if (cid >= 0 && cid < (int)eleCount && 
            std::find(ep_ids.begin(), ep_ids.end(), cid) == ep_ids.end()) {
            ep_ids.push_back(cid);
            searchLayer[cid] = highNode->layer;
            if (ep_ids.size() >= 5) break;
        }
    }

    if (ep_ids.empty() && root->entryPoint >= 0 && root->entryPoint < (int)eleCount) {
        ep_ids.push_back(root->entryPoint);
    }

    ResultHeap result = searchBaseLayer0(ep_ids, vecData, highNode->layer, rangeL, rangeR, ef_s, sp);

    while(result.size() > (size_t)k) result.pop();
    std::priority_queue<std::pair<float, hnswlib::labeltype>> top;
    while(!result.empty()) {
        auto r = result.top();
        result.pop();
        if (r.second >= 0 && r.second < (int)eleCount) {
            top.push({r.first, keyList_[r.second]});
        }
    }
    return top;
}

// ==================== 聚类相关函数实现 ====================

float RangeHNSW::calculateUniqueness(const std::vector<tableint>& points) {
    if (points.size() < 2) return 1.0;
    
    // 随机采样100个点计算平均距离
    int sampleSize = std::min(100, (int)points.size());
    std::vector<int> indices;
    for (int i = 0; i < sampleSize; i++) {
        indices.push_back(rand() % points.size());
    }
    
    float totalDist = 0;
    int count = 0;
    for (int i = 0; i < sampleSize; i++) {
        for (int j = i+1; j < sampleSize; j++) {
            if (points[indices[i]] >= 0 && points[indices[i]] < eleCount &&
                points[indices[j]] >= 0 && points[indices[j]] < eleCount) {
                totalDist += distL2(getVector(points[indices[i]]), 
                                   getVector(points[indices[j]]));
                count++;
            }
        }
    }
    
    float avgDist = count > 0 ? totalDist / count : 0;
    // 归一化到0-1之间（假设最大距离为1000）
    return std::min(1.0f, avgDist / 1000.0f);
}

void RangeHNSW::buildClusterIndex() {
    if (!root) {
        std::cerr << "Error: root is null, cannot build cluster index" << std::endl;
        return;
    }
    
    auto startTime = std::chrono::high_resolution_clock::now();
    
    std::cout << "\n=== Starting Cluster Index Building ===" << std::endl;
    std::cout << "maxLayer = " << maxLayer << std::endl;
    std::cout << "Root layer = " << root->layer << std::endl;
    std::cout << "Total points = " << eleCount << std::endl;
    
    // 动态计算logN大小
    float log2n = std::log2(eleCount);
    size_t baseClusterSize = std::max((size_t)10, (size_t)log2n);
    
    std::cout << "Base cluster size (logN) = " << baseClusterSize << std::endl;
    
    // 分层构建聚类
    std::vector<std::pair<int, int>> levelRanges = {
        {1, 2},   // 低层：大聚类
        {3, 4},   // 中层：标准聚类
        {5, maxLayer}  // 高层：小聚类
    };
    
    for (auto& range : levelRanges) {
        int startLevel = range.first;
        int endLevel = std::min(range.second, maxLayer);
        
        // 根据层数调整聚类大小
        size_t adjustedClusterSize;
        if (startLevel <= 2) {
            adjustedClusterSize = baseClusterSize * 2;  // 低层：大聚类
            std::cout << "\nLow levels " << startLevel << "-" << endLevel 
                      << " using large clusters (size=" << adjustedClusterSize << ")" << std::endl;
        } else if (startLevel <= 4) {
            adjustedClusterSize = baseClusterSize;       // 中层：标准大小
            std::cout << "\nMiddle levels " << startLevel << "-" << endLevel 
                      << " using standard clusters (size=" << adjustedClusterSize << ")" << std::endl;
        } else {
            adjustedClusterSize = std::max((size_t)5, baseClusterSize / 2);   // 高层：小聚类
            std::cout << "\nHigh levels " << startLevel << "-" << endLevel 
                      << " using small clusters (size=" << adjustedClusterSize << ")" << std::endl;
        }
        
        for (int level = startLevel; level <= endLevel; level++) {
            auto levelStart = std::chrono::high_resolution_clock::now();
            
            std::vector<tableint> levelPoints;
            collectLevelPoints(root, level, levelPoints);
            
            if (levelPoints.empty()) {
                std::cout << "Level " << level << " has 0 points" << std::endl;
                continue;
            }
            
            // 计算该层的统计信息
            size_t numPoints = levelPoints.size();
            float uniqueness = calculateUniqueness(levelPoints);
            
            std::cout << "\nLevel " << level << " statistics:" << std::endl;
            std::cout << "  Points: " << numPoints << std::endl;
            std::cout << "  Uniqueness: " << uniqueness << std::endl;
            
            // 根据分布均匀性调整聚类数量
            int numClusters;
            if (uniqueness > 0.8) {
                // 点分布均匀，用标准数量
                numClusters = std::max(2, (int)(numPoints / adjustedClusterSize));
                std::cout << "  Uniform distribution, using standard clustering" << std::endl;
            } else {
                // 点分布不均匀，用更多聚类
                numClusters = std::max(2, (int)(numPoints / (adjustedClusterSize * 0.7)));
                std::cout << "  Skewed distribution, using more clusters" << std::endl;
            }
            
            // 限制聚类数量
            numClusters = std::min(numClusters, (int)(numPoints / 2));
            numClusters = std::max(2, numClusters);
            
            std::cout << "  Creating " << numClusters << " clusters (avg size " 
                      << numPoints/numClusters << ")" << std::endl;
            
            buildClustersForLevel(levelPoints, numClusters, level);
            
            auto levelEnd = std::chrono::high_resolution_clock::now();
            float levelTime = std::chrono::duration<float>(levelEnd - levelStart).count();
            
            // 记录统计信息
            ClusterStats stats;
            stats.level = level;
            stats.numPoints = numPoints;
            stats.numClusters = levelClusters[level].size();
            stats.avgClusterSize = numPoints / (float)stats.numClusters;
            stats.buildTime = levelTime;
            stats.queryHitRate = 0;
            clusterStats.push_back(stats);
        }
    }
    
    auto endTime = std::chrono::high_resolution_clock::now();
    float totalTime = std::chrono::duration<float>(endTime - startTime).count();
    
    // 输出最终统计
    std::cout << "\n=== Cluster Index Summary ===" << std::endl;
    std::cout << "Total clusters: " << clusters.size() << std::endl;
    std::cout << "Build time: " << totalTime << " seconds" << std::endl;
    
    for (auto& stats : clusterStats) {
        std::cout << "Level " << stats.level 
                  << ": " << stats.numClusters << " clusters"
                  << ", avg size " << stats.avgClusterSize
                  << ", build time " << stats.buildTime << "s" << std::endl;
    }
}

void RangeHNSW::buildClustersForLevel(const std::vector<tableint>& points, int k, int level) {
    if (points.empty()) return;
    
    if ((int)points.size() <= k) {
        // 每个点作为一个聚类
        for (auto p : points) {
            if (p >= 0 && p < eleCount) {
                auto cluster = std::make_unique<Cluster>(dim);
                cluster->centroid = getVector(p);
                cluster->representatives.push_back(p);
                cluster->radius = 0;
                cluster->minAttr = cluster->maxAttr = valueList_[p];
                cluster->level = level;
                
                int idx = clusters.size();
                clusters.push_back(std::move(cluster));
                levelClusters[level].push_back(idx);
            }
        }
        return;
    }
    
    // 使用k-means++初始化，但加入智能选择
    std::vector<int> centers;
    
    // 随机选择第一个中心
    std::uniform_int_distribution<> distr(0, points.size() - 1);
    int firstIdx = distr(eng);
    centers.push_back(points[firstIdx]);
    
    // 计算属性值范围
    float attrRange = valueList_[points.back()] - valueList_[points.front()];
    float midAttr = attrRange > 0 ? (valueList_[points.back()] + valueList_[points.front()]) / 2.0 : 0;
    
    std::vector<float> minDists(points.size(), std::numeric_limits<float>::max());
    std::vector<float> attrWeights(points.size(), 1.0);
    
    // 如果属性值范围较大，给远离中位数的点更高权重
    if (attrRange > 0) {
        for (size_t j = 0; j < points.size(); j++) {
            float attrDist = std::abs(valueList_[points[j]] - midAttr) / attrRange;
            attrWeights[j] = 1.0 + attrDist * 0.5;  // 权重范围1.0-1.5
        }
    }
    
    for (int i = 1; i < k; i++) {
        float total = 0;
        
        for (size_t j = 0; j < points.size(); j++) {
            if (points[j] >= 0 && points[j] < eleCount && 
                centers.back() >= 0 && centers.back() < eleCount) {
                
                float vecDist = distL2(getVector(points[j]), getVector(centers.back()));
                minDists[j] = std::min(minDists[j], vecDist);
                total += minDists[j] * attrWeights[j];
            }
        }
        
        if (total <= 0) {
            centers.push_back(points[distr(eng)]);
            continue;
        }
        
        float r = std::uniform_real_distribution<float>(0, total)(eng);
        float sum = 0;
        bool found = false;
        
        for (size_t j = 0; j < points.size(); j++) {
            sum += minDists[j] * attrWeights[j];
            if (sum >= r) {
                centers.push_back(points[j]);
                found = true;
                break;
            }
        }
        
        if (!found) {
            centers.push_back(points.back());
        }
    }
    
    // 确保中心点数量足够
    while ((int)centers.size() < k) {
        centers.push_back(points[distr(eng)]);
    }
    
    // K-means迭代
    std::vector<std::vector<tableint>> assignments(k);
    std::vector<std::vector<float>> centroids(k);
    
    for (int i = 0; i < k; i++) {
        if (centers[i] >= 0 && centers[i] < eleCount) {
            centroids[i] = getVector(centers[i]);
        } else {
            centroids[i].resize(dim, 0);
        }
    }
    
    const int maxIter = 15;
    bool changed = true;
    int iter = 0;
    
    while (changed && iter < maxIter) {
        changed = false;
        iter++;
        
        for (auto& a : assignments) a.clear();
        
        for (auto p : points) {
            if (p < 0 || p >= eleCount) continue;
            
            int best = 0;
            float bestDist = distL2(getVector(p), centroids[0]);
            
            for (int j = 1; j < k; j++) {
                float d = distL2(getVector(p), centroids[j]);
                if (d < bestDist) {
                    bestDist = d;
                    best = j;
                }
            }
            assignments[best].push_back(p);
        }
        
        for (int j = 0; j < k; j++) {
            if (assignments[j].size() < 2) continue;
            
            std::vector<float> newCentroid(dim, 0);
            for (auto p : assignments[j]) {
                if (p >= 0 && p < eleCount) {
                    auto vec = getVector(p);
                    for (int d = 0; d < dim; d++) {
                        newCentroid[d] += vec[d];
                    }
                }
            }
            
            float invSize = 1.0f / assignments[j].size();
            for (int d = 0; d < dim; d++) {
                newCentroid[d] *= invSize;
            }
            
            float diff = 0;
            for (int d = 0; d < dim; d++) {
                diff += (newCentroid[d] - centroids[j][d]) * (newCentroid[d] - centroids[j][d]);
            }
            
            if (diff > 1e-3) {
                changed = true;
                centroids[j] = newCentroid;
            }
        }
    }
    
    // 创建最终的聚类
    int clustersCreated = 0;
    for (int j = 0; j < k; j++) {
        if (assignments[j].size() < 2) continue;
        
        auto cluster = std::make_unique<Cluster>(dim);
        cluster->centroid = centroids[j];
        
        float maxDist = 0;
        std::vector<std::pair<float, tableint>> dists;
        std::vector<std::pair<int, tableint>> attrSorted;
        
        for (auto p : assignments[j]) {
            if (p < 0 || p >= eleCount) continue;
            
            float d = distL2(getVector(p), centroids[j]);
            if (d > maxDist) maxDist = d;
            dists.push_back({d, p});
            attrSorted.push_back({valueList_[p], p});
            
            if (valueList_[p] < cluster->minAttr) cluster->minAttr = valueList_[p];
            if (valueList_[p] > cluster->maxAttr) cluster->maxAttr = valueList_[p];
        }
        
        cluster->radius = maxDist;
        
        // 选择代表点：距离质心最近的2个点
        std::sort(dists.begin(), dists.end());
        for (int r = 0; r < std::min(2, (int)dists.size()); r++) {
            cluster->representatives.push_back(dists[r].second);
        }
        
        // 如果聚类足够大，也添加属性值中位数附近的点
        if (assignments[j].size() > 5) {
            std::sort(attrSorted.begin(), attrSorted.end());
            int midIdx = attrSorted.size() / 2;
            if (std::find(cluster->representatives.begin(), 
                         cluster->representatives.end(), 
                         attrSorted[midIdx].second) == cluster->representatives.end()) {
                cluster->representatives.push_back(attrSorted[midIdx].second);
            }
        }
        
        cluster->level = level;
        
        int idx = clusters.size();
        clusters.push_back(std::move(cluster));
        levelClusters[level].push_back(idx);
        clustersCreated++;
    }
    
    if (clustersCreated == 0) {
        std::cout << "Warning: No clusters created for level " << level << std::endl;
    }
}

std::vector<int> RangeHNSW::kmeansPlusPlusInit(const std::vector<tableint>& points, int k) {
    std::vector<int> centers;
    std::vector<float> minDists(points.size(), std::numeric_limits<float>::max());
    
    if (points.empty()) return centers;
    
    std::uniform_int_distribution<> distr(0, points.size() - 1);
    int firstIdx = distr(eng);
    centers.push_back(points[firstIdx]);
    
    for (int i = 1; i < k; i++) {
        float total = 0;
        
        for (size_t j = 0; j < points.size(); j++) {
            if (points[j] >= 0 && points[j] < eleCount && 
                centers.back() >= 0 && centers.back() < eleCount) {
                
                float d = distL2(getVector(points[j]), getVector(centers.back()));
                minDists[j] = std::min(minDists[j], d);
                total += minDists[j];
            }
        }
        
        if (total <= 0) {
            centers.push_back(points[distr(eng)]);
            continue;
        }
        
        float r = std::uniform_real_distribution<float>(0, total)(eng);
        float sum = 0;
        bool found = false;
        
        for (size_t j = 0; j < points.size(); j++) {
            sum += minDists[j];
            if (sum >= r) {
                centers.push_back(points[j]);
                found = true;
                break;
            }
        }
        
        if (!found) {
            centers.push_back(points.back());
        }
    }
    
    return centers;
}

std::vector<tableint> RangeHNSW::findClusterCandidates(float* queryVec, int rangeL, int rangeR, node* targetNode) {
    std::vector<tableint> candidates;
    if (clusters.empty() || !targetNode) {
        return candidates;
    }
    
    totalQueries++;
    
    int startLevel = targetNode->layer;
    if (startLevel < 0 || startLevel > maxLayer) startLevel = maxLayer;
    
    std::unordered_set<tableint> seen;
    std::vector<std::pair<float, tableint>> scoredCandidates;
    
    // 多尺度搜索：从当前层向上搜索3层
    int searchLevels = std::min(3, maxLayer - startLevel + 1);
    
    for (int offset = 0; offset < searchLevels; offset++) {
        int level = startLevel + offset;
        if (level > maxLayer) break;
        
        auto it = levelClusters.find(level);
        if (it == levelClusters.end()) continue;
        
        // 找到该层最相关的聚类
        std::vector<std::pair<float, int>> clusterScores;
        
        for (int idx : it->second) {
            if (idx < 0 || idx >= (int)clusters.size()) continue;
            auto& cluster = clusters[idx];
            
            // 快速属性过滤
            if (cluster->minAttr > rangeR || cluster->maxAttr < rangeL)
                continue;
            
            // 计算距离评分
            float d = distL2(queryVec, cluster->centroid.data());
            float score = d / (cluster->radius + 1e-6);
            clusterScores.push_back({score, idx});
        }
        
        // 按评分排序
        std::sort(clusterScores.begin(), clusterScores.end());
        
        // 选择最好的几个聚类
        int numToTake = std::min(2, (int)clusterScores.size());
        for (int i = 0; i < numToTake; i++) {
            int idx = clusterScores[i].second;
            auto& cluster = clusters[idx];
            
            // 如果评分足够好，加入代表点
            if (clusterScores[i].first <= 3.0) {
                for (auto rep : cluster->representatives) {
                    if (rep >= 0 && rep < eleCount && 
                        seen.find(rep) == seen.end() && 
                        valueList_[rep] >= rangeL && 
                        valueList_[rep] <= rangeR) {
                        
                        float repDist = distL2(queryVec, getVector(rep).data());
                        float repScore = repDist / (cluster->radius + 1e-6);
                        
                        scoredCandidates.push_back({repScore, rep});
                        seen.insert(rep);
                    }
                }
                
                // 记录命中
                levelHitCount[level]++;
            }
        }
    }
    
    // 按质量评分排序，只保留最好的几个
    std::sort(scoredCandidates.begin(), scoredCandidates.end());
    for (int i = 0; i < std::min(3, (int)scoredCandidates.size()); i++) {
        candidates.push_back(scoredCandidates[i].second);
    }
    
    // 每1000次查询输出一次命中率统计
    if (totalQueries % 1000 == 0) {
        std::cout << "\n=== Cluster Hit Rate Statistics (Query " << totalQueries << ") ===" << std::endl;
        for (auto& hit : levelHitCount) {
            float hitRate = hit.second / (float)totalQueries;
            std::cout << "Level " << hit.first << " hit rate: " << hitRate << std::endl;
            
            // 更新统计信息中的命中率
            for (auto& stats : clusterStats) {
                if (stats.level == hit.first) {
                    stats.queryHitRate = hitRate;
                    break;
                }
            }
        }
    }
    
    return candidates;
}

void RangeHNSW::collectLevelPoints(RangeHNSW::node* nd, int targetLevel, std::vector<tableint>& points) {
    if (!nd) return;
    
    if (nd->layer == targetLevel) {
        std::vector<tableint> nodePoints;
        traverse(nodePoints, nd);
        points.insert(points.end(), nodePoints.begin(), nodePoints.end());
    } else if (nd->layer > targetLevel) {
        for (int i = 0; i <= nd->keynum; i++) {
            if (nd->child[i]) {
                collectLevelPoints(nd->child[i], targetLevel, points);
            }
        }
    }
}

void RangeHNSW::validateClusterIndex() {
    std::cout << "\n=== Cluster Index Validation ===" << std::endl;
    
    int totalClusters = 0;
    for (auto& levelClust : levelClusters) {
        totalClusters += levelClust.second.size();
    }
    
    std::cout << "Total clusters in levelClusters: " << totalClusters << std::endl;
    std::cout << "Total clusters in clusters vector: " << clusters.size() << std::endl;
    
    if (totalClusters != (int)clusters.size()) {
        std::cerr << "Warning: Cluster count mismatch!" << std::endl;
    }
    
    // 验证每个聚类的数据
    int validClusters = 0;
    int totalRepresentatives = 0;
    for (auto& cluster : clusters) {
        if (cluster && !cluster->representatives.empty()) {
            validClusters++;
            totalRepresentatives += cluster->representatives.size();
        }
    }
    std::cout << "Valid clusters: " << validClusters << std::endl;
    std::cout << "Average representatives per cluster: " 
              << (validClusters ? totalRepresentatives/(float)validClusters : 0) << std::endl;
}

// ==================== B树相关辅助函数 ====================

bool RangeHNSW::cmp(int a, int b) {
    if (a < 0 || a >= (int)eleCount || b < 0 || b >= (int)eleCount) return a < b;
    if(valueList_[a] != valueList_[b]) return valueList_[a] < valueList_[b];
    return keyList_[a] < keyList_[b];
}

void RangeHNSW::traverse(std::vector<tableint>& result, RangeHNSW::node* nd) {
    if (!nd) return;
    
    if (nd->layer == 0) {
        if (nd->entryPoint >= 0 && nd->entryPoint < (int)eleCount) {
            result.push_back(nd->entryPoint);
        }
        return;
    }
    for(int i = 0; i <= nd->keynum; i++) {
        if (nd->child[i]) {
            traverse(result, nd->child[i]);
        }
    }
}

int RangeHNSW::findRight(RangeHNSW::node* nd) {
    if (!nd) return -1;
    if(nd->layer == 0) return nd->entryPoint;
    if (nd->child[nd->keynum]) {
        return findRight(nd->child[nd->keynum]);
    }
    return -1;
}

RangeHNSW::node* RangeHNSW::findHighNode(RangeHNSW::node* node, int rangeL, int rangeR) {
    if (!node) return nullptr;
    if(node->layer == 0) return node;
    
    int belongL = node->keynum;
    int belongR = node->keynum;
    
    for(int i = 0; i < node->keynum; i++) {
        if (node->key[i] >= 0 && node->key[i] < (int)eleCount) {
            if(rangeL < valueList_[node->key[i]] ||
               (rangeL == valueList_[node->key[i]] && node->child[i] && 
                rangeL == valueList_[findRight(node->child[i])])) {
                belongL = i;
                break;
            }
        }
    }
    for(int i = 0; i < node->keynum; i++) {
        if (node->key[i] >= 0 && node->key[i] < (int)eleCount) {
            if(rangeR < valueList_[node->key[i]]) {
                belongR = i;
                break;
            }
        }
    }
    
    belongL = std::max(0, std::min(belongL, node->keynum));
    belongR = std::max(0, std::min(belongR, node->keynum));
    
    if(belongL == belongR && node->child[belongL]) {
        return findHighNode(node->child[belongL], rangeL, rangeR);
    }
    return node;
}

int RangeHNSW::findPosition(RangeHNSW::node* nd, int value) {
    if (!nd) return 0;
    
    for(int i = 0; i < nd->keynum; i++) {
        if (nd->key[i] >= 0 && nd->key[i] < (int)eleCount) {
            if(value < valueList_[nd->key[i]]) {
                return i;
            }
        }
    }
    return nd->keynum;
}

// ==================== 图搜索相关函数 ====================

tableint RangeHNSW::findEntry(const void *query_data, RangeHNSW::node *nd, tableint currObj) const {
    if (!nd || currObj < 0 || currObj >= eleCount) return currObj;
    
    float curdist = fstdistfunc_(query_data, getDataByInternalId(currObj), dist_func_param_);
    int endLayer = nd->layer;
    int startLayer = findEntryLayer(endLayer);
    
    for (int layer = startLayer; layer < endLayer; layer += skipLayer) {
        bool changed = true;
        while (changed) {
            changed = false;
            
            for(int l = 0; l <= 0; l++) {
                unsigned int *data = (unsigned int *)get_linklist(currObj, layer - l);
                if (!data) continue;
                
                int size = getListCount(data);
                if (size <= 0) continue;
                
                tableint *datal = (tableint *)(data + 1);
                for (int i = 0; i < size; i++) {
                    tableint cand = datal[i];
                    if (cand < 0 || cand >= eleCount) continue;
                    
                    float d = fstdistfunc_(query_data, getDataByInternalId(cand), dist_func_param_);
                    
                    if (d < curdist) {
                        curdist = d;
                        currObj = cand;
                        changed = true;
                    }
                }
            }
        }
    }
    return currObj;
}

int RangeHNSW::findEntryLayer(int Layer) const {
    return Layer % skipLayer;
}

RangeHNSW::ResultHeap RangeHNSW::searchBaseLayer0(
        std::vector<tableint> ep_ids, 
        const void *data_point, 
        int Layer, 
        int rangeL, 
        int rangeR, 
        int ef, 
        int splitPoint) {
    
    tag++;
    if (tag == 0) tag = 1;
    
    ResultHeap top_candidates;
    ResultHeap candidateSet;
    
    float lowerBound;
    
    for (auto ep_id : ep_ids) {
        if (ep_id < 0 || ep_id >= eleCount || isDeleted[ep_id]) continue;
        
        float dist = fstdistfunc_(data_point, getDataByInternalId(ep_id), dist_func_param_);
        candidateSet.emplace(-dist, ep_id);
        visited_array[ep_id] = tag;
        searchLayer[ep_id] = Layer;
        
        if (valueList_[ep_id] >= rangeL && valueList_[ep_id] <= rangeR) {
            top_candidates.emplace(dist, ep_id);
        }
    }
    
    lowerBound = top_candidates.empty() ? std::numeric_limits<float>::max() : top_candidates.top().first;
    
    while (!candidateSet.empty()) {
        auto curr = candidateSet.top();
        candidateSet.pop();
        
        tableint curId = curr.second;
        if (curId < 0 || curId >= eleCount) continue;
        
        short int layer = searchLayer[curId];
        
        for (int l = 0; l <= 1; l++) {
            if (layer - l < 0) continue;
            
            auto* ll = get_linklist(curId, layer - l);
            if (!ll) continue;
            
            int size = getListCount(ll);
            if (size <= 0) continue;
            
            auto* neighbors = (tableint*)(ll + 1);
            
            for (int j = 0; j < size; j++) {
                tableint nid = neighbors[j];
                if (nid < 0 || nid >= eleCount) continue;
                if (visited_array[nid] == tag) continue;
                
                visited_array[nid] = tag;
                
                float dist = fstdistfunc_(data_point, getDataByInternalId(nid), dist_func_param_);
                
                if ((int)top_candidates.size() < ef || lowerBound > dist) {
                    candidateSet.emplace(-dist, nid);
                    searchLayer[nid] = layer;
                    
                    if (!isDeleted[nid] && valueList_[nid] >= rangeL && valueList_[nid] <= rangeR) {
                        top_candidates.emplace(dist, nid);
                        
                        if ((int)top_candidates.size() > ef) {
                            top_candidates.pop();
                        }
                        
                        if (!top_candidates.empty()) {
                            lowerBound = top_candidates.top().first;
                        }
                    }
                }
            }
        }
        
        if (splitPoint != -1) {
            auto* ll = get_linklist(curId, Layer);
            if (!ll) continue;
            
            int size = getListCount(ll);
            if (size <= 0) continue;
            
            auto* neighbors = (tableint*)(ll + 1);
            
            for (int j = 0; j < size; j++) {
                tableint nid = neighbors[j];
                if (nid < 0 || nid >= eleCount) continue;
                if (visited_array[nid] == tag) continue;
                if (valueList_[nid] < rangeL || valueList_[nid] > rangeR) continue;
                
                visited_array[nid] = tag;
                
                float dist = fstdistfunc_(data_point, getDataByInternalId(nid), dist_func_param_);
                
                if ((int)top_candidates.size() < ef || lowerBound > dist) {
                    candidateSet.emplace(-dist, nid);
                    searchLayer[nid] = layer;
                    
                    top_candidates.emplace(dist, nid);
                    
                    if ((int)top_candidates.size() > ef) {
                        top_candidates.pop();
                    }
                    
                    if (!top_candidates.empty()) {
                        lowerBound = top_candidates.top().first;
                    }
                }
            }
        }
    }
    
    return top_candidates;
}

void RangeHNSW::getNeighborsByHeuristic2(ResultHeap &top_candidates, const size_t M) {
    if (top_candidates.size() < M) return;
    
    std::priority_queue<std::pair<float, tableint>> queue_closest;
    std::vector<std::pair<float, tableint>> return_list;
    while (top_candidates.size() > 0) {
        queue_closest.emplace(-top_candidates.top().first, top_candidates.top().second);
        top_candidates.pop();
    }
    
    while (queue_closest.size()) {
        if (return_list.size() >= M) break;
        std::pair<float, tableint> curent_pair = queue_closest.top();
        float dist_to_query = -curent_pair.first;
        queue_closest.pop();
        bool good = true;
        
        for (std::pair<float, tableint> second_pair : return_list) {
            float curdist = fstdistfunc_(getDataByInternalId(second_pair.second),
                                         getDataByInternalId(curent_pair.second),
                                         dist_func_param_);
            if (curdist < dist_to_query) {
                good = false;
                break;
            }
        }
        if (good) {
            return_list.push_back(curent_pair);
        }
    }
    
    for (std::pair<float, tableint> curent_pair : return_list) {
        top_candidates.emplace(-curent_pair.first, curent_pair.second);
    }
}

RangeHNSW::ResultHeap RangeHNSW::searchBaseLayer(
        const std::vector<tableint> &ep_ids, 
        const void *data_point, 
        int layer) {
    
    tag++;
    if (tag == 0) tag = 1;
    
    ResultHeap top_candidates;
    ResultHeap candidateSet;
    
    float lowerBound;
    
    for (auto ep_id : ep_ids) {
        if (ep_id < 0 || ep_id >= eleCount || isDeleted[ep_id]) continue;
        
        float dist = fstdistfunc_(data_point, getDataByInternalId(ep_id), dist_func_param_);
        top_candidates.emplace(dist, ep_id);
        candidateSet.emplace(-dist, ep_id);
        visited_array[ep_id] = tag;
    }
    
    lowerBound = top_candidates.empty() ? std::numeric_limits<float>::max() : top_candidates.top().first;
    
    while (!candidateSet.empty()) {
        auto curr = candidateSet.top();
        if ((-curr.first) > lowerBound && (int)top_candidates.size() >= ef_construction) {
            break;
        }
        candidateSet.pop();
        
        tableint curId = curr.second;
        if (curId < 0 || curId >= eleCount) continue;
        
        for (int i = 0; i <= 0; i++) {
            if (layer - i < 0) break;
            
            auto* ll = get_linklist(curId, layer - i);
            if (!ll) continue;
            
            int size = getListCount(ll);
            if (size <= 0) continue;
            
            auto* neighbors = (tableint*)(ll + 1);
            
            for (int j = 0; j < size; j++) {
                tableint nid = neighbors[j];
                if (nid < 0 || nid >= eleCount) continue;
                if (visited_array[nid] == tag) continue;
                
                visited_array[nid] = tag;
                
                float dist = fstdistfunc_(data_point, getDataByInternalId(nid), dist_func_param_);
                
                if ((int)top_candidates.size() < ef_construction || lowerBound > dist) {
                    candidateSet.emplace(-dist, nid);
                    
                    if (!isDeleted[nid]) {
                        top_candidates.emplace(dist, nid);
                        if ((int)top_candidates.size() > ef_construction) {
                            top_candidates.pop();
                        }
                        if (!top_candidates.empty()) {
                            lowerBound = top_candidates.top().first;
                        }
                    }
                }
            }
        }
    }
    
    return top_candidates;
}

// ==================== 辅助函数 ====================

inline char *RangeHNSW::getDataByInternalId(tableint internal_id) const {
    if (internal_id < 0 || internal_id >= eleCount) return nullptr;
    return (char*)(vecData_ + internal_id * data_size_);
}

linklistsizeint *RangeHNSW::get_linklist(tableint internal_id, int layer) const {
    if (internal_id < 0 || internal_id >= linklist.size()) return nullptr;
    if (layer < 0 || layer > maxLayer) return nullptr;
    if (!linklist[internal_id]) return nullptr;
    
    return (linklistsizeint *)(linklist[internal_id] + sizeLinkList * layer);
}

unsigned short int RangeHNSW::getListCount(linklistsizeint *ptr) const {
    if (!ptr) return 0;
    return *((unsigned short int *)ptr);
}

void RangeHNSW::setListCount(linklistsizeint *ptr, unsigned short int size) const {
    if (!ptr) return;
    *((unsigned short int*)(ptr)) = *((unsigned short int *)&size);
}

std::vector<float> RangeHNSW::getVector(tableint id) const {
    std::vector<float> vec(dim, 0.0f);
    if (id < 0 || id >= eleCount) return vec;
    
    float* data = (float*)(vecData_ + id * data_size_);
    memcpy(vec.data(), data, dim * sizeof(float));
    return vec;
}

float RangeHNSW::distL2(const float* a, const float* b) const {
    if (!a || !b) return std::numeric_limits<float>::max();
    
    float dist = 0;
    for (int i = 0; i < dim; i++) {
        float diff = a[i] - b[i];
        dist += diff * diff;
    }
    return dist;
}

float RangeHNSW::distL2(const std::vector<float>& a, const std::vector<float>& b) const {

    }

}
